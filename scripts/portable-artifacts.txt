# portable-artifacts.txt — 便携运行时制品清单（**唯一事实来源**）
#
# 谁读它：`scripts/portable-runtime.sh`（纯 bash 解析，不依赖任何解释器，保证"冷启动
# 的机器上也能先把 Node 装进仓库"）。
#
# 纪律：
#   * **sha256 必须来自官方发布物**（nodejs 的 SHASUMS256.txt / GitHub Release 的 digest），
#     禁止凭记忆或"看起来像"填写；换版本时连同 sha256 一起更新，并跑
#     `bash scripts/portable-runtime.sh verify` 自证。
#   * 本文件只描述"怎么拿到运行时"，**不放二进制**（`runtime/` 整目录 gitignore）。
#
# 格式：
#   `键=值`   —— 头部参数（镜像、版本、pip 源）
#   `kind<TAB>platform<TAB>文件名<TAB>sha256` —— 制品行；platform 与 `pi_platform` 同名，
#                                               也允许 `any`（所有平台通用）

# ── Node：官方 tarball，镜像优先（本机实测 nodejs.org 在本网络下 25 MB 要 >10 分钟，
#          阿里云镜像 500 KB/s 约 50 秒；两者内容与官方 SHASUMS 一致）────────────
node_version=22.23.1
node_min=22.19.0
node_mirrors=https://mirrors.aliyun.com/nodejs-release https://nodejs.org/dist

# ── Python：python-build-standalone 的 install_only_stripped（含 pip，开箱可用）──
# 为什么不复用系统 python3：把包装进 `--target` 目录虽然可搬迁，但 C 扩展（Pillow、
# pymupdf…）的 ABI 绑死解释器小版本；换台机器 python 版本不同就得重装。自带解释器后
# "跟着项目走"才真正成立。
python_version=3.12.15
python_release=20261013
python_url=https://github.com/astral-sh/python-build-standalone/releases/download/{release}/{file}

# GitHub Release 代理兜底（本机实测：直连 github.com/objects.githubusercontent.com 时通时断，
# `gh-proxy.com` ≈316 KB/s、`ghproxy.net` ≈115 KB/s 可用；`ghfast.top`/`mirror.ghproxy.com` 不通）。
# **安全前提**：无论走谁，下载后都要过 sha256（下面的制品行），代理只能原样转发或失败，
# 不能替换内容 —— 否则校验会失败并丢弃。故代理只影响"快不快"，不影响"对不对"。
github_proxies=https://gh-proxy.com/ https://ghproxy.net/

# ── SearXNG：web_search 的本地后端（**源码也放仓库内**，含它自己的 Python 依赖）──
# 为什么用 commit 而是 main：`codeload .../tar.gz/<commit>` 的内容地址不可变，
# 于是"锚定 + sha256"两重保证；`main` 会随时变，无法复核。
searxng_commit=b00212a8366131936856b726ed4dd9bd281e4d7e
searxng_url=https://codeload.github.com/searxng/searxng/tar.gz/{commit}

# ── pip 源（国内直连 pypi.org 慢；与 .npmrc 的 npmmirror 同思路）─────────────
pip_index=https://mirrors.aliyun.com/pypi/simple/
pip_host=mirrors.aliyun.com
